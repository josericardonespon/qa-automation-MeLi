        function reorderAuthFooterButtons() {

            const authFooter = document.querySelector('.auth-footer');

            if (authFooter) {

                let primaryButton = null;

                let secondaryButton = null;

                

                primaryButton = authFooter.querySelector('.cancelb') || 

                               authFooter.querySelector('[class*="cancelb"]') ||

                               document.querySelector('.cancelb') || 

                               document.querySelector('[class*="cancelb"]');

                

                if (!primaryButton) {

                    primaryButton = authFooter.querySelector('.js-cancel');

                }

                

                secondaryButton = authFooter.querySelector('.js-switchAuthenticator') || 

                                document.querySelector('.js-switchAuthenticator');

                

                if (!secondaryButton && primaryButton && primaryButton.classList.contains('cancelb')) {

                    secondaryButton = authFooter.querySelector('.js-cancel:not(.cancelb)') ||

                                    authFooter.querySelector('.js-cancel:not([class*="cancelb"])') ||

                                    document.querySelector('.js-cancel:not(.cancelb)') ||

                                    document.querySelector('.js-cancel:not([class*="cancelb"])');

                }

                

                if (primaryButton && secondaryButton) {

                    if (!authFooter.contains(primaryButton)) {

                        const primaryParent = primaryButton.parentNode;

                        primaryParent.removeChild(primaryButton);

                        authFooter.appendChild(primaryButton);

                    }

                    

                    if (!authFooter.contains(secondaryButton)) {

                        const secondaryParent = secondaryButton.parentNode;

                        secondaryParent.removeChild(secondaryButton);

                        authFooter.appendChild(secondaryButton);

                    }

                    

                    if (authFooter.firstChild !== primaryButton) {

                        authFooter.insertBefore(primaryButton, authFooter.firstChild);

                    }

                    

                    if (primaryButton.nextSibling !== secondaryButton) {

                        authFooter.insertBefore(secondaryButton, primaryButton.nextSibling);

                    }

                }

                else if (primaryButton && !secondaryButton) {

                    if (!authFooter.contains(primaryButton)) {

                        const primaryParent = primaryButton.parentNode;

                        primaryParent.removeChild(primaryButton);

                        authFooter.appendChild(primaryButton);

                    }

                }

            }

        }

function checkErrors() {

    const observerTarget = document.body;

    const observer = new MutationObserver(function(mutationsList, observer) {

        const errorElement = document.querySelector('.o-form-error-container.o-form-has-errors');

        if (errorElement) {

            const errorDiv = errorElement.querySelector('.okta-form-infobox-error');

            if (errorDiv) {

                const pElement = errorDiv.querySelector('p');

                setLoginerrorMsg(pElement);

                observer.disconnect();

            }

            errorElement.style.display = '';

        }

    });



    const observerConfig = {

        childList: true,

        subtree: true

    };



    observer.observe(observerTarget, observerConfig);

}

        function checkStaticUserNotAssignedError() {

    const errorTextPor = "O usuário não está atribuído ao aplicativo.";

    const errorTextEn = "User is not assigned to this application.";

    

    const walker = document.createTreeWalker(

        document.body,

        NodeFilter.SHOW_TEXT,

        null,

        false

    );

    

    let textNode;

    while (textNode = walker.nextNode()) {

        const text = textNode.textContent.trim();

        if (text === errorText || text === errorTextPor || text === errorTextEn) {

            

            let errorContainer = textNode.parentElement;

            while (errorContainer && !errorContainer.classList.contains('o-form-error-container')) {

                errorContainer = errorContainer.parentElement;

            }

            

            if (errorContainer) {

                const pElement = errorContainer.querySelector('.infobox-error div') || 

                               errorContainer.querySelector('.infobox-error p') ||

                               textNode.parentElement;

                

                if (pElement) {

                    if (es) {

        }

function checkIdentifyErrors(isSecurityAccessDenied = false, isUserNotAssigned = false, isUnauthorizedError = false, isUnknownError = false, isInsufficientFactorCount = false) {

    const primaryAuthScreen = document.querySelector('.siw-main-view.identify.primary-auth');

    

    if (!primaryAuthScreen) {

        return;

    }

    

    const errorContainer = document.querySelector('.o-form-error-container') || 

                          document.querySelector('[data-se="o-form-error-container"]');

    

    if (!errorContainer) {

        return;

    const primaryAuthObserver = new MutationObserver(function(mutationsList, observer) {

        mutationsList.forEach((mutation) => {

            if (mutation.type === 'childList') {

                mutation.addedNodes.forEach((node) => {

                    if (node.nodeType === Node.ELEMENT_NODE) {

                        const isP = node.tagName === 'P';

                        const containsP = node.querySelector && node.querySelector('p');

                        

                        if (isP || containsP) {

                            const pElement = isP ? node : containsP;

                                }

                            }

                    }

            }

        });

        childList: true,

        subtree: true,
